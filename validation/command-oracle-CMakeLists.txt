cmake_minimum_required(VERSION 3.8)
project(prometheus_command_oracle)
find_package(ament_cmake REQUIRED)
find_package(prometheus_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(command_oracle prometheus_command_oracle.cpp)
target_compile_features(command_oracle PRIVATE cxx_std_17)
target_compile_options(command_oracle PRIVATE -O2 -fno-fast-math)
target_link_libraries(command_oracle Eigen3::Eigen)
ament_target_dependencies(command_oracle prometheus_msgs)
